12/// Supplies the value exported for an OQueue message when using [`register_with`](super::registry::register_with).
13///
14/// The `DefaultProjection` derive recursively projects fields that implement this trait. POD values
15/// project to themselves; mark fields that should be opaque identifiers with
16/// `#[projection(opaque)]`, which projects them through [`OpaqueId`], or omit fields with
17/// `#[projection(omit)]`. POD values that cannot implement `Projectable` (due to orphan rules) can
18/// be annotated with `#[projection(self)]` to produce the same behavior.
19#[diagnostic::on_unimplemented(
20    message = "`{Self}` does not implement `DefaultProjection`",
21    note = "derive `DefaultProjection` for a local type, or use `#[projection(opaque)]` on a field that should be represented by an ID;",
22    note = "for a Copy, Send, and Serialize type that projects to itself, derive `Projectable`"
23)]
24pub trait DefaultProjection {
25    /// The projected value type written to the export stream.
26    type Projection: Copy + Send + Serialize + 'static;
27
28    /// Returns the default projection of this message.
29    fn default_projection(&self) -> Self::Projection;
30}
31
32/// Tag to mark types as trivially projectable
33#[diagnostic::on_unimplemented(
34    message = "`{Self}` does not implement `Projectable`",
35    note = "derive `Projectable` if this type can be copied, sent, and serialized as-is"
36)]
37pub trait Projectable {}

85macro_rules! impl_tuple_default_projection {
86    ($(($($ty:ident),+)),+ $(,)?) => {
87        $(impl<$($ty: DefaultProjection),+> DefaultProjection for ($($ty,)+) {
88            type Projection = ($($ty::Projection,)+);
89
90            #[expect(non_snake_case)]
91            fn default_projection(&self) -> Self::Projection {
92                let ($($ty,)+) = self;
93                ($($ty.default_projection(),)+)
94            }
95        })+
96    };
97}

118impl<T> DefaultProjection for T
119where
120    T: Copy + Send + Serialize + Projectable + 'static,
121{
122    type Projection = Self;
123
124    fn default_projection(&self) -> Self::Projection {
125        *self
126    }
127}
128
129impl<T: DefaultProjection> DefaultProjection for Option<T> {
130    type Projection = Option<T::Projection>;
131
132    fn default_projection(&self) -> Self::Projection {
133        self.as_ref().map(DefaultProjection::default_projection)
134    }
135}
136
137impl<T: DefaultProjection, E: DefaultProjection> DefaultProjection for Result<T, E> {
138    type Projection = Result<T::Projection, E::Projection>;
139
140    fn default_projection(&self) -> Self::Projection {
141        match self {
142            Ok(value) => Ok(value.default_projection()),
143            Err(error) => Err(error.default_projection()),
144        }
145    }
146}

148/// An ID of an object. This include an opaque ID (often a pointer value), and the type name. The
149/// type name is stored as a reference to a static string, so this struct is small and `Copy`.
150#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
151pub struct OID {
152    /// An opaque ID of the object. This value has no meaning, but is unique for the lifetime of the
153    /// object.
154    ///
155    /// NOTE: Ideally, this would be unique over the lifetime of the system. However, that isn't
156    /// true yet.
157    pub id: usize,
158
159    /// The name of the type of the object. This comes from [`core::any::type_name`] and, like that,
160    /// has an undefined format, so it cannot safely be parsed.
161    pub type_name: &'static str,
162}
163
164impl OID {
165    /// Create the OID for a value. `T` should be provided explicitly because doing so forces the
166    /// compiler to generate the ID from something of type `T`. That prevents accidentally
167    /// generating the ID from a pointer to a reference or a box-like type. The compiler auto-deref
168    /// rules will generally unwrap the object as needed to get the correct reference.
169    pub fn of_ref<T: ?Sized>(v: &T) -> OID {
170        OID {
171            id: core::ptr::from_ref(v) as *const () as usize,
172            type_name: type_name::<T>(),
173        }
174    }
175}
176
177/// Supplies an opaque identifier for a value projected with `#[projection(opaque)]`.
178///
179/// By default, the identifier is the address of the value itself along with the name of the type.
180/// Smart pointers and references identify the value they point to, so that copies of the same smart
181/// pointer share an identifier.
182pub trait OpaqueId {
183    /// Returns the opaque identifier of this value.
184    fn opaque_id(&self) -> OID;
185}
186
187impl<T: ?Sized> OpaqueId for &T {
188    fn opaque_id(&self) -> OID {
189        OID::of_ref::<T>(self)
190    }
191}
192
193impl<T: ?Sized> OpaqueId for Arc<T> {
194    fn opaque_id(&self) -> OID {
195        OID::of_ref::<T>(self)
196    }
197}
198
199impl<T: ?Sized> OpaqueId for Box<T> {
200    fn opaque_id(&self) -> OID {
201        OID::of_ref::<T>(self)
202    }
203}

277    #[ktest]
278    fn default_projection_serializes_nested_custom_and_opaque_fields() {
279        let message = ProjectedMessage {
280            leaf: ProjectedLeaf { value: 0x1234_5678 },
281            ref_type: "opaque",
282            special: SpecialProjection(41),
283        };
284        let projected = message.default_projection();
285
286        let mut actual = Vec::new();
287        Serialize::serialize(
288            &projected,
289            &mut minicbor_serde::Serializer::new(&mut actual),
290        )
291        .unwrap();
292
293        let mut decoder = minicbor::Decoder::new(&actual);
294        assert_eq!(decoder.map().unwrap(), Some(3));
295        assert_eq!(decoder.str().unwrap(), "leaf");
296        assert_eq!(decoder.map().unwrap(), Some(1));
297        assert_eq!(decoder.str().unwrap(), "value");
298        assert_eq!(decoder.u32().unwrap(), message.leaf.value);
299        assert_eq!(decoder.str().unwrap(), "ref_type");
300        assert_eq!(decoder.map().unwrap(), Some(2));
301        assert_eq!(decoder.str().unwrap(), "id");
302        decoder.u64().unwrap();
303        assert_eq!(decoder.str().unwrap(), "type_name");
304        assert!(decoder.str().unwrap().contains("str"));
305        assert_eq!(decoder.str().unwrap(), "special");
306        assert_eq!(decoder.u32().unwrap(), 42);
307        assert_eq!(decoder.position(), actual.len());
308    }
