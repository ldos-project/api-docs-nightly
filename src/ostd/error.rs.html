17/// The trait of all errors carrying OSTD metadata.
18///
19/// This specifically provides access to an [`StackInfo`]. Implement this using the
20/// [`ostd_error`] attribute macro.
21pub trait OstdError: snafu::Error {
22    /// Get the context captured with the error.
23    fn stack_info(&self) -> Option<&StackInfo>;
24}
25
26/// The error type which is returned from the APIs of this crate.
27#[ostd_error]
28#[derive(Snafu, Debug)]
29#[snafu(visibility(pub))]
30pub enum Error {
31    /// Invalid arguments provided.
32    #[snafu(display("Invalid arguments ({context})"))]
33    InvalidArgs,
34    /// Insufficient memory available.
35    #[snafu(display("Insufficient memory ({context})"))]
36    NoMemory,
37    /// Page fault occurred.
38    #[snafu(display("Page fault ({context})"))]
39    PageFault,
40    /// Access to a resource is denied.
41    #[snafu(display("Access denied ({context})"))]
42    AccessDenied,
43    /// Input/output error.
44    #[snafu(display("Input/output error ({context})"))]
45    IoError,
46    /// Insufficient system resources.
47    #[snafu(display("Insufficient resources ({context})"))]
48    NotEnoughResources,
49    /// Arithmetic Overflow occurred.
50    #[snafu(display("Arithmetic overflow ({context})"))]
51    Overflow,
52    /// An OQueue error
53    #[snafu(transparent)]
54    #[ostd(context(source))]
55    #[cfg(not(baseline_asterinas))]
56    OQueueError {
57        /// Source OQueue error
58        source: OQueueError,
59    },
60}
61
62/// The error kind and captured context.
63#[cfg(not(baseline_asterinas))]
64#[derive(Clone, Copy, Serialize)]
65pub struct ErrorDefaultProjection {
66    kind: &'static str,
67    context: Option<StackInfoDefaultProjection>,
68}
69
70#[cfg(not(baseline_asterinas))]
71impl DefaultProjection for Error {
72    type Projection = ErrorDefaultProjection;
73
74    fn default_projection(&self) -> Self::Projection {
75        let kind = match self {
76            Self::InvalidArgs { .. } => "InvalidArgs",
77            Self::NoMemory { .. } => "NoMemory",
78            Self::PageFault { .. } => "PageFault",
79            Self::AccessDenied { .. } => "AccessDenied",
80            Self::IoError { .. } => "IoError",
81            Self::NotEnoughResources { .. } => "NotEnoughResources",
82            Self::Overflow { .. } => "Overflow",
83            #[cfg(not(baseline_asterinas))]
84            Self::OQueueError { .. } => "OQueueError",
85        };
86
87        ErrorDefaultProjection {
88            kind,
89            context: self.stack_info().map(DefaultProjection::default_projection),
90        }
91    }
92}
