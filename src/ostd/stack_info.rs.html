19/// A location in the code. Unlike, [`snafu::Location`] and [`core::panic::Location`], this can hold
20/// a dynamic string. However, if given a `&'static str`, it will not allocate memory (internally it
21/// uses [`Cow`]).
22#[derive(Clone, Debug)]
23pub struct Location {
24    /// The file where the error was reported.
25    ///
26    /// This can be either a `&'static str` or a `String`.
27    pub file: Cow<'static, str>,
28    /// The line where the error was reported
29    pub line: u32,
30    /// The column where the error was reported
31    pub column: u32,
32}
33
34/// The source location, with a filename when it has static storage.
35#[cfg(not(baseline_asterinas))]
36#[derive(Clone, Copy, Debug, Eq, PartialEq, Serialize)]
37pub struct LocationDefaultProjection {
38    file: Option<&'static str>,
39    line: u32,
40    column: u32,
41}
42
43#[cfg(not(baseline_asterinas))]
44impl DefaultProjection for Location {
45    type Projection = LocationDefaultProjection;
46
47    fn default_projection(&self) -> Self::Projection {
48        LocationDefaultProjection {
49            file: match &self.file {
50                Cow::Borrowed(file) => Some(*file),
51                Cow::Owned(_) => None,
52            },
53            line: self.line,
54            column: self.column,
55        }
56    }
57}
58
59impl Display for Location {
60    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
61        write!(
62            f,
63            "{file}:{line}:{column}",
64            file = self.file,
65            line = self.line,
66            column = self.column,
67        )
68    }
69}

71impl From<&snafu::Location> for Location {
72    fn from(value: &snafu::Location) -> Self {
73        Self {
74            file: value.file.into(),
75            line: value.line,
76            column: value.column,
77        }
78    }
79}

91impl Location {
92    /// Create a [`Location`] from a [`core::panic::Location`] with a limited lifetime. This
93    /// allocates memory in which to store the filename.
94    pub fn from_location(value: &core::panic::Location<'_>) -> Self {
95        Self {
96            file: value.file().to_owned().into(),
97            line: value.line(),
98            column: value.column(),
99        }
100    }

103/// The stack context in which an error may have occurred. This should be used in most error types.
104///
105/// This type may also be useful for capturing the context of some other event for use in later
106/// errors or tracing. For example, a lock might capture the context every time it is locked, so
107/// that `try_lock` errors can include the lock holder context along with `try_lock` call context.
108/// This can be very useful for understanding deadlocks.
109///
110/// NOTE: This module has some degree of "premature" optimization. This is because we want to
111/// encourage [`StackInfo`] to be used liberally so keeping the cost as low as possible has a
112/// very significant value.
113#[derive(Clone, Debug)]
114#[non_exhaustive]
115pub struct StackInfo {
116    /// The stacktrace.
117    pub stack_trace: CapturedStackTrace,
118    /// The ID of the running [`Task`].
119    pub task_id: Option<NonZeroUsize>,
120    /// The ID of a CPU which was being executed on around the time the context was captured. This
121    /// may or may not be the CPU on which the actual error occurred as a migration could happen
122    /// during context capture.
123    pub cpu_id: CpuId,
124    /// The source location.
125    pub location: Option<Location>,
126}
127
128/// The task, CPU, and source location captured with an error.
129#[cfg(not(baseline_asterinas))]
130#[derive(Clone, Copy, Debug, Eq, PartialEq, Serialize)]
131pub struct StackInfoDefaultProjection {
132    task_id: Option<NonZeroUsize>,
133    cpu_id: u32,
134    location: Option<LocationDefaultProjection>,
135}
136
137#[cfg(not(baseline_asterinas))]
138impl DefaultProjection for StackInfo {
139    type Projection = StackInfoDefaultProjection;
140
141    fn default_projection(&self) -> Self::Projection {
142        StackInfoDefaultProjection {
143            task_id: self.task_id,
144            cpu_id: self.cpu_id.into(),
145            location: self.location.default_projection(),
146        }
147    }
148}
149
150impl StackInfo {
151    /// Create a new `StackInfo`, dropping `skip` stackframes starting from the caller to this
152    /// function.
153    #[track_caller]
154    pub fn new(skip: usize) -> StackInfo {
155        let current_task = Task::current();
156        let task_id = current_task.as_ref().map(|t| t.id());
157        let stack_trace = CapturedStackTrace::capture(skip + 1);
158        StackInfo {
159            stack_trace,
160            task_id,
161            cpu_id: CpuId::current_racy(),
162            location: Some(core::panic::Location::caller().into()),
163        }
164    }
165
166    /// Create an "empty" [`StackInfo`] which contains all default information.
167    pub const fn empty() -> Self {
168        Self {
169            stack_trace: CapturedStackTrace::empty(),
170            task_id: None,
171            cpu_id: CpuId::bsp(),
172            location: None,
173        }
174    }
175}
176
177impl Display for StackInfo {
178    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
179        if let Some(location) = &self.location {
180            write!(f, "at {location} ")?;
181        }
182        if let Some(tid) = self.task_id {
183            write!(f, "on task {tid} ")?;
184        }
185        write!(f, "on cpu {}", self.cpu_id.as_usize())?;
186        write!(f, " with stack: {}", self.stack_trace)?;
187        Ok(())
188    }
189}
190
191impl GenerateImplicitData for StackInfo {
192    #[track_caller]
193    fn generate() -> Self {
194        // We drop 2 frames: one for this call to `generate`, and one for the error constructor
195        // (such as, a snafu generated `into_error` method).
196        StackInfo::new(2)
197    }
198}
