14/// A struct signifying the kernel is within an fpu section
15/// Enables nested sections and dropping the last nested one ensures that the user fpu state is
16/// restored if applicable
17pub struct InKernelFpuGuard {
18    task: CurrentTask,
19    #[cfg(debug_assertions)]
20    expected_depth: usize,
21    _not_send: PhantomData<*const ()>,
22}
23
24impl Drop for InKernelFpuGuard {
25    fn drop(&mut self) {
26        assert!(self.task.fpu_section_depth() > 0);
27        #[cfg(debug_assertions)]
28        debug_assert_eq!(self.task.fpu_section_depth(), self.expected_depth);
29        let depth = self.task.fpu_section_depth() - 1;
30        self.task.set_fpu_section_depth(depth);
31    }
32}
33
34/// Start a kernel fpu section and ensure that the following invariants are true:
35/// - The kernel is not servicing an interrupt
36/// - The user fpu state is saved
37/// - Nested kernel fpu regions do not accidentally clobber each other
38/// - In debug mode kernel fpu sections should resolve in the order they were created.
39pub fn enable_kernel_fpu() -> InKernelFpuGuard {
40    assert!(InterruptLevel::current().is_task_context());
41    let task = Task::current().expect("enable_kernel_fpu called without a current task");
42
43    task.with_kernel_fpu_context(|kernel_context| {
44        kernel_context.get_or_insert_with(|| Box::new(FpuContext::new()));
45    });
46
47    let outermost = task.fpu_section_depth() == 0;
48    if outermost {
49        if crate::task::try_with_current_fpu_context(|context| context.enter_kernel()).is_none() {
50            task.enter_kernel();
51        }
52        init_kernel_fpu_control_registers();
53    }
54    let expected_depth = task.fpu_section_depth() + 1;
55    task.set_fpu_section_depth(expected_depth);
56
57    InKernelFpuGuard {
58        task,
59        #[cfg(debug_assertions)]
60        expected_depth,
61        _not_send: PhantomData,
62    }
63}
64
65/// Used to ensure that the kernel fpu sections save their state before a context switch
66pub(crate) fn before_switch(_next_task: &Task) {
67    if let Some(current_task) = Task::current()
68        && current_task.fpu_section_depth() > 0
69    {
70        current_task.with_kernel_fpu_context(|context| {
71            context.as_mut().unwrap().save();
72        });
73    }
74}

76#[cfg(ktest)]
77mod tests {
78    use alloc::sync::Arc;
79    use core::{
80        arch::asm,
81        sync::atomic::{AtomicBool, Ordering},
82    };
83
84    use super::enable_kernel_fpu;
85    use crate::{
86        prelude::ktest,
87        task::{Task, TaskOptions},

89
90    #[ktest]
91    fn nested_sections_preserve_depth() {
92        let input = [1.5_f64, 2.5];
93        let outer_value = [3.5_f64, 4.5];
94        let inner_value = [5.5_f64, 6.5];
95        let mut output = [0_f64; 2];
96        unsafe { asm!("movupd xmm15, [{}]", in(reg) input.as_ptr(), options(nostack)) };
97        let outer = enable_kernel_fpu();
98        assert_eq!(Task::current().unwrap().fpu_section_depth(), 1);
99        unsafe { asm!("movupd xmm15, [{}]", in(reg) outer_value.as_ptr(), options(nostack)) };
100
101        let inner = enable_kernel_fpu();
102        assert_eq!(Task::current().unwrap().fpu_section_depth(), 2);
103        unsafe { asm!("movupd xmm15, [{}]", in(reg) inner_value.as_ptr(), options(nostack)) };
104        drop(inner);
105
106        unsafe { asm!("movupd [{}], xmm15", in(reg) output.as_mut_ptr(), options(nostack)) };
107        assert_eq!(output, inner_value);
108        assert_eq!(Task::current().unwrap().fpu_section_depth(), 1);
109        drop(outer);
110
111        unsafe { asm!("movupd [{}], xmm15", in(reg) output.as_mut_ptr(), options(nostack)) };
112        assert_eq!(output, inner_value);
113        assert_eq!(Task::current().unwrap().fpu_section_depth(), 0);
114    }
115
116    #[ktest]
117    fn active_section_survives_task_switch() {
118        let input = [7.5_f64, 8.5];
119        let task_value = [9.5_f64, 10.5];
120        let mut output = [0_f64; 2];
121        let switched = Arc::new(AtomicBool::new(false));
122        let switched_clone = switched.clone();
123
124        unsafe { asm!("movupd xmm15, [{}]", in(reg) input.as_ptr(), options(nostack)) };
125        let section = enable_kernel_fpu();
126        unsafe { asm!("movupd xmm15, [{}]", in(reg) input.as_ptr(), options(nostack)) };
127
128        let task = Arc::new(
129            TaskOptions::new(move || {
130                unsafe {
131                    asm!("movupd xmm15, [{}]", in(reg) task_value.as_ptr(), options(nostack))
132                };
133                switched_clone.store(true, Ordering::Release);
134                Task::yield_now();
135            })
136            .build()
137            .unwrap(),
138        );
139        task.run();
140
141        while !switched.load(Ordering::Acquire) {
142            Task::yield_now();
143        }
144
145        unsafe { asm!("movupd [{}], xmm15", in(reg) output.as_mut_ptr(), options(nostack)) };
146        assert_eq!(output, input);
147        drop(section);
148    }
