3//! A common framework for minibenchmarks in Mariposa.
4//!
5//! A benchmark supplies a closure that measures one iteration; [`run`] collects a sample from it per
6//! iteration and captures them, and [`report`] emits the metadata needed to interpret the result.
7
8use alloc::{boxed::Box, sync::Arc, vec::Vec};
9use core::fmt::Debug;
10
11use aster_logger::println;
12// Capture to a block device is built on OQueues, so it does not exist in baseline mode.
13#[cfg(not(baseline_asterinas))]
14use mariposa_data_capture::DataCaptureFile;
15use ostd::{arch::tsc_freq, ostd_error};
16use serde::Serialize;
17use snafu::Snafu;
18
19/// Prefix on every line a benchmark prints, so a host tool can pick the block out of a console log.
20/// A benchmark reporting why it gave up prints `{PREFIX}|error <name>: <reason>`.
21pub(crate) const PREFIX: &str = "MARIPOSA_BENCH";
22
23#[ostd_error]
24#[derive(Debug, Snafu)]
25pub enum Error {
26    #[snafu(display("cannot reserve room for {iterations} samples ({context})"))]
27    OutOfMemory { iterations: usize },
28}
29
30/// Collects one sample per iteration for `iterations` iterations and write the results to a capture file
31/// when the benchmarking is finished, returning how many were captured.
32///
33/// `benchmark_fn` is given the iteration's sequence number and returns that iteration's sample.
34#[cfg(not(baseline_asterinas))]
35pub fn run<S, E>(
36    iterations: u32,
37    capture_file: Arc<DataCaptureFile<S>>,
38    mut benchmark_fn: impl FnMut(u64) -> Result<S, E>,
39) -> Result<usize, E>
40where
41    S: Copy + Send + Serialize + 'static,
42    E: From<Error>,
43{
44    // Reserve the whole sample array up front so recording never allocates on the hot path.
45    let mut samples: Vec<S> = Vec::new();
46    if samples.try_reserve_exact(iterations as usize).is_err() {
47        return Err(OutOfMemorySnafu {
48            iterations: iterations as usize,
49        }
50        .build()
51        .into());
52    }
53
54    // The benchmarking loop.
55    for seq in 0..iterations as u64 {
56        samples.push(benchmark_fn(seq)?);
57    }
58
59    let measured = samples.len();
60    capture_file.write_values(Box::new(samples.into_iter()));
61    // Stopping the capture file syncs it and waits for the worker to exit, so the
62    // samples are on the device before this returns and the caller reports success.
63    capture_file.stop();
64
65    Ok(measured)
66}
67
68/// Emits the metadata needed to interpret a completed run: the benchmark's name, the TSC frequency
69/// its cycle counts are relative to, the configuration it ran under, and how many samples it
70/// captured.
71pub fn report(name: &str, config: &dyn Debug, samples: usize) {
72    println!("{PREFIX}|begin {name}");
73    println!("{PREFIX}|tsc_freq_hz {}", tsc_freq());
74    println!("{PREFIX}|config {config:?}");
75    println!("{PREFIX}|samples {samples}");
76    println!("{PREFIX}|end {name}");
77}
