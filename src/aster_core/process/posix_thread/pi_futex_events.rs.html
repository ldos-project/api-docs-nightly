3//! The `futex.pi.events` OQueue, which reports how threads contend on PI futexes.
4//!
5//! Together with `scheduler.events`, this lets a policy see who waits on whom. Every event carries
6//! an optional snapshot of the current owner and its scheduling policy. `Block` details name the
7//! waiter and its policy, while `WaitEnd` and `Handoff` details report how long the wait lasted
8//! and who the lock went to.
9//!
10//! Event delivery never waits for observers. If a strong observer's ring is
11//! full, the event is dropped. The queue is exported at `/oqueues/futex/pi/events`.
12
13use serde::Serialize;
14
15use crate::{
16    prelude::*,
17    process::Pid,
18    sched::SchedPolicyView,
19    thread::{Thread, Tid},
20    time::clocks::MonotonicClock,
21};
22
23/// An event on a PI futex.
24#[derive(Clone, Copy, Debug, Serialize)]
25pub(crate) struct PiFutexEvent {
26    /// The monotonic time of the event, in nanoseconds.
27    pub time_ns: u64,
28    /// The process whose address space holds the futex.
29    pub pid: Pid,
30    /// The user-space address of the futex word.
31    pub futex_addr: u64,
32    /// The current owner sampled while preparing the event, not the original blocking owner.
33    ///
34    /// `None` means the futex is unowned or a consistent owner snapshot was unavailable (for
35    /// example, the word was unmapped, the owner exited, or ownership changed during sampling).
36    /// Ownership and policy can change again before the event is published or observed.
37    pub owner: Option<PiFutexOwner>,
38    pub details: PiFutexEventDetails,
39}
40
41/// A snapshot of a PI futex owner's identity and scheduling policy.
42#[derive(Clone, Copy, Debug, Serialize)]
43pub(crate) struct PiFutexOwner {
44    pub tid: Tid,
45    pub policy: SchedPolicyView,
46}
47
48/// The details specific to a PI futex event.
49#[derive(Clone, Copy, Debug, Serialize)]
50pub(crate) enum PiFutexEventDetails {
51    /// `waiter` started to wait for the futex.
52    Block {
53        waiter: Tid,
54        waiter_policy: SchedPolicyView,
55    },
56    /// `waiter` stopped waiting after `waited_ns` nanoseconds.
57    WaitEnd {
58        waiter: Tid,
59        waited_ns: u64,
60        outcome: PiWaitOutcome,
61    },
62    /// `from` released the futex directly to the waiter `to`, leaving `waiters_left` other
63    /// threads waiting.
64    Handoff {
65        from: Tid,
66        to: Tid,
67        waiters_left: u32,
68    },
69}
70
71/// Why a PI futex waiter stopped waiting.
72#[derive(Clone, Copy, Debug, Serialize)]
73pub(crate) enum PiWaitOutcome {
74    /// The waiter was handed the lock.
75    Acquired,
76    /// The waiter was woken without the lock (e.g., the owner died) and will try again.
77    Retry,
78    TimedOut,
79    Interrupted,
80}
81
82impl PiFutexEvent {
83    pub(crate) fn new(
84        futex_addr: Vaddr,
85        owner: Option<PiFutexOwner>,
86        details: PiFutexEventDetails,
87    ) -> Self {
88        Self {
89            time_ns: now_ns(),
90            pid: current!().pid(),
91            futex_addr: futex_addr as u64,
92            owner,
93            details,
94        }
95    }
96}
97
98/// Returns the current monotonic time in nanoseconds.
99pub(crate) fn now_ns() -> u64 {
100    MonotonicClock::get().read_time().as_nanos() as u64
101}
102
103pub(crate) fn policy_of(thread: &Thread) -> SchedPolicyView {
104    thread.sched_attr().policy().into()
105}

107// TODO: This module follows the pattern used for `scheduler.events` and `lockprio.decisions`: a
108// `Once` holding the producer, an `init` that creates and registers the OQueue, a `produce`
109// wrapper, and a no-op copy of both for `baseline_asterinas`. Factor this into a general
110// structure, perhaps with utilities to create it, and use it consistently instead of repeating it.
111#[cfg(not(baseline_asterinas))]
112mod imp {
113    use ostd::{
114        orpc::oqueue::{GenericOQueueRef, OQueue as _, OQueueBase as _, RefProducer, registry},
115        path,
116    };
117    use spin::Once;
118
119    use super::PiFutexEvent;
120
121    static PRODUCER: Once<RefProducer<PiFutexEvent>> = Once::new();
122
123    pub(crate) fn init() {
124        PRODUCER.call_once(|| {
125            let path = path!(futex.pi.events);
126            let oqueue = GenericOQueueRef::new(1024, path.clone());
127            registry::register(&path, &oqueue.as_any_oqueue());
128            oqueue
129                .attach_ref_producer()
130                .expect("a new OQueue always allows a ref producer")
131        });
132    }
133
134    pub(crate) fn produce(event: impl FnOnce() -> PiFutexEvent) {
135        if let Some(producer) = PRODUCER.get() {
136            // A slow observer must not block a thread holding a futex bucket lock.
137            // TODO(aneesh): use either revocable observers or only allow weak observers
138            let _ = producer.try_produce_ref(&event());
139        }
140    }
141}
