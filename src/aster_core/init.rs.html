48pub(super) fn main() {
49    // TODO(amp, BIG MERGE): Reconsider this initialization order, and placement of initialization code.
50
51    // CRITICAL: Initialize scheduler and thread spawning BEFORE component initialization!
52    // - Scheduler must be injected first, otherwise OSTD's fallback FIFO scheduler gets used
53    // - The thread spawn function must be injected, otherwise threads are created without
54    //   Thread association and are ignored by the ClassScheduler
55    thread::init();
56    sched::init();
57
58    // Initialize the global states for all CPUs.
59    ostd::early_println!("OSTD initialized. Preparing components.");
60    component::init_all(InitStage::Bootstrap, component::parse_metadata!()).unwrap();
61    sched::init_select_cpu_policy();
62    init();
63
64    // Initialize the per-CPU states for BSP.
65    init_on_each_cpu();
66
67    // Enable APs.
68    ostd::boot::smp::register_ap_entry(ap_init);
69
70    // Give the control of the BSP to the idle thread.
71    ThreadOptions::new(bsp_idle_loop)
72        .cpu_affinity(CpuId::bsp().into())
73        .sched_policy(SchedPolicy::Idle)
74        .spawn();
75}

95/// Initialize kernel subsystems for ktests. This is not a complete initialization, but does load
96/// all components.
97///
98/// This is idempotent and should be called in *every* test that needs it. This avoids at least some
99/// test order dependence.
100///
101/// TODO(arthurp, https://github.com/ldos-project/asterinas/issues/221): Something less ad-hoc.
102#[cfg(ktest)]
103pub(crate) fn init_for_ktest() {
104    static INITIALIZED: Once<()> = Once::new();
105
106    INITIALIZED.call_once(|| {
107        component::init_all(InitStage::Bootstrap, component::parse_metadata!()).unwrap();
108        crate::time::init();
109        vm::vmar::init_in_first_kthread();
110    });
111}

120fn ap_init() {
121    // Initialize the per-CPU states for AP.
122    init_on_each_cpu();
123
124    ThreadOptions::new(ap_idle_loop)
125        // No races because `ap_init` runs on a certain AP.
126        .cpu_affinity(CpuId::current_racy().into())
127        .sched_policy(SchedPolicy::Idle)
128        .spawn();
129}
130
131//--------------------------------------------------------------------------
132// Per-CPU idle threads
133//--------------------------------------------------------------------------
134
135// Note: Keep the code in the idle loop to the bare minimum.
136//
137// We do not want the idle loop to
138// rely on the APIs of other kernel subsystems for two reasons.
139// First, the idle task must never sleep or block.
140// This property is relied upon by the scheduler.
141// Second, the idle task is spawned before the kernel is fully initialized.
142// So other subsystems may not be ready, yet.
143//
144// In addition,
145// doing more work in the idle task may have negative impact on
146// the latency to switching from the idle task to a useful, runnable one.
147
148fn bsp_idle_loop() {
149    ostd::info!("Idle thread for CPU #0 started");
150
151    // Spawn the first non-idle kernel thread on BSP.
152    ThreadOptions::new(first_kthread)
153        .cpu_affinity(CpuId::bsp().into())
154        .sched_policy(SchedPolicy::default())
155        .spawn();
156
157    // Wait till the init process is spawned.
158    let init_process = loop {
159        if let Some(init_process) = INIT_PROCESS.get() {
160            break init_process;
161        };
162
163        ostd::task::halt_cpu();
164    };
165
166    // Wait till the init process becomes zombie.
167    while !init_process.status().is_zombie() {
168        ostd::task::halt_cpu();
169    }
170
171    // TODO(amp, BIG MERGE): This may or may not run depending on how shutdown is now handled.
172    #[cfg(not(baseline_asterinas))]
173    for f in data_capture::DATA_CAPTURE_FILE_FINALIZERS.lock().drain(..) {
174        f();
175    }
176
177    panic!(
178        "The init process terminates with code {:?}",
179        init_process.status().exit_code()
180    );
181}

183fn ap_idle_loop() {
184    ostd::info!(
185        "Idle thread for CPU #{} started",
186        // No races because this function runs on a certain AP.
187        CpuId::current_racy().as_usize(),
188    );
189
190    loop {
191        ostd::task::halt_cpu();
192    }
193}

199// The main function of the first (non-idle) kernel thread
200fn first_kthread() {
201    println!("Spawn the first kernel thread");
202
203    let init_mnt_ns = MountNamespace::get_init_singleton();
204    let fs_resolver = init_mnt_ns.new_path_resolver();
205    init_in_first_kthread(&fs_resolver);
206    let boot_init = prepare_boot_init(fs_resolver);
207
208    print_banner();
209
210    // Run benchmarks when bench.benchmark is set in the kernel args
211    if benchmarks::is_benchmark_enabled() {
212        benchmarks::BenchmarkHarness::run();
213        poweroff(ExitCode::Success);
214    }
215
216    INIT_PROCESS.call_once(|| {
217        let karg = INIT_PROC_ARGS.get().unwrap();
218        let argv = karg.argv().to_vec();
219        let envp = karg.envp().to_vec();
220        boot_init
221            .spawn(argv, envp)
222            .expect("failed to run the init process")
223    });
224
225    #[cfg(not(baseline_asterinas))]
226    benchmarks::oqueue_roundtrip::init_after_init_process();
227
228    #[cfg(not(baseline_asterinas))]
229    oqfs_test::init_after_init_process();
230
231    #[cfg(not(baseline_asterinas))]
232    if VM_HUGEPAGED_ENABLED.load(Ordering::Relaxed) {
233        vm::hugepaged::spawn(INIT_PROCESS.get().expect("initialed already").clone());
234    }
235
236    #[cfg(target_arch = "x86_64")]
237    #[cfg(not(baseline_asterinas))]
238    if PMU_DTLB_ENABLED.load(Ordering::Relaxed) {
239        let pmu = crate::arch::pmu::PmuServer::spawn();
240        pmu.reset();
241        pmu.start();
242    }
243
244    #[cfg(not(baseline_asterinas))]
245    if SCHEDULER_CAPTURE_DATA.load(Ordering::Relaxed) {
246        use ostd::{
247            orpc::oqueue::LifetimelessElementDescriptor, task::scheduler::info::CommonSchedInfo,
248        };
249
250        if let Some(oqueue) = lookup_by_path::<LifetimelessElementDescriptor<SchedulingEvent>>(
251            &path!(scheduler.events),
252        ) {
253            #[derive(Debug, Clone, Copy, Serialize)]
254            struct KernelSchedulingEvent {
255                timestamp: Instant,
256                task: TaskId,
257                kind: SchedulingEventKind,
258                /// The CPU whose run queue holds the task, or `None` once it has gone to sleep.
259                /// For a `Schedule` event, this is the CPU about to run the task.
260                cpu: Option<u32>,
261            }
262
263            let capture_file = new_data_capture_file::<KernelSchedulingEvent>(
264                mariposa_data_capture::FileDescriptor {
265                    path: path!(scheduler.events),
266                    length: 500 * 1024 * 1024,
267                },
268            );
269
270            if let Some(capture_file) = capture_file {
271                capture_file.register_observer(ObserverRegistration {
272                    path: path!(scheduler.events),
273                    observer: oqueue
274                        .attach_strong_observer(Projection::new(|e: &SchedulingEvent| {
275                            let context = EventContext::new();
276                            KernelSchedulingEvent {
277                                timestamp: context.timestamp,
278                                task: TaskId::new(&e.task),
279                                kind: e.kind,
280                                cpu: e.task.cpu().get().map(|cpu| cpu.as_usize() as u32),
281                            }
282                        }))
283                        .unwrap(),
284                });
285                capture_file.start();
286            }
287        } else {
288            error!("Could not find scheduler.events OQueue. Scheduler events will not be captured.")
289        }
290    }
291
292    #[cfg(not(baseline_asterinas))]
293    if IO_CAPTURE_BLOCK_IO.load(Ordering::Relaxed) {
294        use aster_block::bio::{BlockDeviceCompletionStats, SubmittedBio};
295
296        use crate::data_capture::new_data_capture_data_file_by_type;
297
298        #[derive(Clone, Copy, Serialize)]
299        struct SubmittedBioEvent {
300            byte_range: (usize, usize),
301            timestamp: Option<Instant>,
302            context: EventContext,
303        }
304        new_data_capture_data_file_by_type::<SubmittedBio, _>(
305            path!(io.block.submitted),
306            500 * 1024 * 1024,
307            || {
308                Projection::new(|e: &SubmittedBio| {
309                    let sid_range = e.sid_range();
310                    let context = EventContext::new();
311                    SubmittedBioEvent {
312                        byte_range: (sid_range.start.to_offset(), sid_range.end.to_offset()),
313                        timestamp: e.submission_time().map(|t| t.into()),
314                        context,
315                    }
316                })
317            },
318        );
319
320        #[derive(Clone, Copy, Serialize)]
321        struct BlockDeviceCompletionEvent {
322            stats: BlockDeviceCompletionStats,
323            context: EventContext,
324        }
325        new_data_capture_data_file_by_type::<BlockDeviceCompletionStats, _>(
326            path!(io.block.completion),
327            500 * 1024 * 1024,
328            || {
329                Projection::new(|stats| {
330                    let context = EventContext::new();
331                    BlockDeviceCompletionEvent {
332                        stats: *stats,
333                        context,
334                    }
335                })
336            },
337        );
338    }
339}

341struct BootInit {
342    path_resolver: PathResolver,
343    init_path: Option<(Path, &'static str)>,
344}
345
346#[repr(u8)]
347enum BootSource {
348    Initramfs = 0,
349    Rootfs = 1,
350}
351
352pub(crate) fn booted_from_rootfs() -> bool {
353    BOOT_SOURCE.load(Ordering::Relaxed) == BootSource::Rootfs as u8
354}
355
356impl BootInit {
357    fn spawn(self, argv: Vec<CString>, envp: Vec<CString>) -> Result<Arc<Process>> {
358        let Self {
359            path_resolver,
360            init_path,
361        } = self;
362
363        // Only rootfs boot may have no init path; initramfs boot always provides one.
364        let Some((init_path, init_name)) = init_path else {
365            return spawn_default_rootfs_init(path_resolver, argv, envp);
366        };
367
368        println!("[kernel] running {} as the init process", init_name);
369        spawn_init_process(
370            path_resolver,
371            init_path,
372            with_init_argv0(init_name, argv),
373            envp,
374        )
375    }
376}
377
378fn prepare_boot_init(mut path_resolver: PathResolver) -> BootInit {
379    if let Ok(init_path) = initramfs::find_init(&path_resolver) {
380        BOOT_SOURCE.store(BootSource::Initramfs as u8, Ordering::Relaxed);
381        return BootInit {
382            path_resolver,
383            init_path: Some(init_path),
384        };
385    }
386
387    rootfs::switch_to_rootfs(&mut path_resolver)
388        .expect("neither an initramfs init nor a usable root filesystem was available");
389    let init_path = rootfs::find_init(&path_resolver).expect("failed to resolve rootfs init path");
390    BOOT_SOURCE.store(BootSource::Rootfs as u8, Ordering::Relaxed);
391    BootInit {
392        path_resolver,
393        init_path,
394    }
395}
396
397fn spawn_default_rootfs_init(
398    path_resolver: PathResolver,
399    argv: Vec<CString>,
400    envp: Vec<CString>,
401) -> Result<Arc<Process>> {
402    // Linux probes the fallback init executables in this order:
403    // Reference: <https://elixir.bootlin.com/linux/v6.19/source/init/main.c#L1634>.
404    const DEFAULT_INIT_EXEC_PATHS: &[&str] = &["/sbin/init", "/etc/init", "/bin/init", "/bin/sh"];
405
406    let mut last_error = None;
407
408    for &init_name in DEFAULT_INIT_EXEC_PATHS {
409        // FIXME: Avoid cloning `argv` and `envp` for each fallback candidate.
410        let init_path = match path_resolver.lookup(&FsPath::try_from(init_name).unwrap()) {
411            Ok(path) => path,
412            Err(error) => {
413                last_error = Some(error);
414                continue;
415            }
416        };
417
418        match spawn_init_process(
419            path_resolver.clone(),
420            init_path,
421            with_init_argv0(init_name, argv.clone()),
422            envp.clone(),
423        ) {
424            Ok(process) => {
425                println!("[kernel] running {} as the rootfs init", init_name);
426                return Ok(process);
427            }
428            Err(error) => last_error = Some(error),
429        }
430    }
431
432    Err(last_error.unwrap())
433}
434
435fn with_init_argv0(init_name: &str, mut argv: Vec<CString>) -> Vec<CString> {
436    // Linux prepends the init executable path as `argv[0]`.
437    // Reference: <https://elixir.bootlin.com/linux/v6.19/source/init/main.c#L1491>.
438    argv.insert(0, CString::new(init_name).unwrap());
439    argv
440}

448fn init_in_first_kthread(path_resolver: &PathResolver) {
449    component::init_all(InitStage::Kthread, component::parse_metadata!()).unwrap();
450    // Initializes the work queue and the other kernel threads' state. This must happen
451    // before interrupts are enabled, in case any IRQ handler uses the work queue as a
452    // bottom half.
453    thread::init_in_first_kthread();
454    crate::fs::init_in_first_kthread(path_resolver);
455
456    #[cfg(not(baseline_asterinas))]
457    {
458        data_capture::start_capture_devices();
459    }
460
461    crate::device::init_in_first_kthread();
462    crate::net::init_in_first_kthread();
463    #[cfg(not(target_arch = "loongarch64"))]
464    crate::vdso::init_in_first_kthread();
465    vm::vmar::init_in_first_kthread();
466    crate::syscall::init_in_first_kthread();
467}
