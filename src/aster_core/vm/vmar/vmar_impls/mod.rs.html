48/// The VMAR (used to be Virtual Memory Address Region, but now an orphan
49/// initialism).
50///
51/// A VMAR is the address space of a process.
52pub(crate) struct Vmar {
53    /// VMAR inner
54    inner: RwMutex<VmarInner>,
55    /// The attached `VmSpace`
56    vm_space: Arc<VmSpace>,
57    /// The RSS counters
58    rss_counters: [PerCpuCounter; NUM_RSS_COUNTERS],
59    /// The process VM
60    process_vm: ProcessVm,
61    /// The number of handles that this `Vmar` has (see [`super::VmarHandle`])
62    num_handles: AtomicUsize,
63    /// Weak self reference
64    weak_self: Weak<Self>,
65
66    rss_hwm_counters: [CpuLocalBox<AtomicIsize>; NUM_RSS_COUNTERS],
67}
68
69impl Vmar {
70    /// Creates a new VMAR.
71    ///
72    /// This method should only be invoked by [`super::VmarHandle`].
73    pub(super) fn new(process_vm: ProcessVm) -> Arc<Self> {
74        let inner = VmarInner::new();
75        let mut vm_space = VmSpace::new();
76        if huge_mapping_enabled() {
77            vm_space = vm_space.with_mapping_policy(Arc::new(VmMappingPolicyGreedyHugeMapping {}));
78        }
79        let rss_counters = array::from_fn(|_| PerCpuCounter::new());
80        let rss_hwm_counters =
81            array::from_fn(|_| alloc_cpu_local(|_| AtomicIsize::new(0)).unwrap());
82        Arc::new_cyclic(move |weak_self| Vmar {
83            inner: RwMutex::new(inner),
84            vm_space: Arc::new(vm_space),
85            rss_counters,
86            process_vm,
87            num_handles: AtomicUsize::new(1),
88            weak_self: weak_self.clone(),
89            rss_hwm_counters,
90        })
91    }
92
93    /// Returns the current RSS count for the given RSS type.
94    pub(crate) fn get_rss_counter(&self, rss_type: RssType) -> usize {
95        self.rss_counters[rss_type as usize].sum_all_cpus()
96    }
97
98    pub(crate) fn get_rss_hwm_counter(&self, rss_type: RssType) -> usize {
99        let mut total: isize = 0;
100        for cpu in all_cpus() {
101            total = total.wrapping_add(
102                self.rss_hwm_counters[rss_type as usize]
103                    .get_on_cpu(cpu)
104                    .load(Ordering::Relaxed),
105            );
106        }
107        if total < 0 {
108            // The counter is unsigned. But an observer may see a negative
109            // value due to race conditions. We return zero if it happens.
110            0
111        } else {
112            total as usize
113        }
114    }
115
116    /// Returns the total size of the mappings in bytes.
117    pub(crate) fn get_mappings_total_size(&self) -> usize {
118        self.inner.read().total_vm
119    }
120
121    /// Returns the attached `VmSpace`.
122    pub(crate) fn vm_space(&self) -> &Arc<VmSpace> {
123        &self.vm_space
124    }
125
126    /// Returns the attached `ProcessVm`.
127    pub(crate) fn process_vm(&self) -> &ProcessVm {
128        &self.process_vm
129    }
130
131    /// Returns whether this VMAR has multiple handles.
132    pub(crate) fn has_multiple_handles(&self) -> bool {
133        self.num_handles.load(Ordering::Relaxed) > 1
134    }
135
136    /// Increases the number of handles.
137    ///
138    /// This method should only be invoked by [`super::VmarHandle`].
139    pub(super) fn inc_num_handles(&self) {
140        let old_num_handles = self.num_handles.fetch_add(1, Ordering::Relaxed);
141        debug_assert_ne!(old_num_handles, 0);
142    }
143
144    /// Decreases the number of handles.
145    ///
146    /// This method should only be invoked by [`super::VmarHandle`].
147    pub(super) fn dec_num_handles(&self) {
148        let old_num_handles = self.num_handles.fetch_sub(1, Ordering::Relaxed);
149        debug_assert_ne!(old_num_handles, 0);
150        if old_num_handles == 1 {
151            // Clear all the mappings. The last process using this VMAR exited
152            // or executed a new program, so this VMAR no longer has a handle.
153            self.clear();
154        }
155    }
156
157    fn add_rss_counter(&self, rss_type: RssType, val: isize) {
158        // There are races but updating a remote counter won't cause any problems.
159        let cpu_id = CpuId::current_racy();
160        let rss = self.rss_counters[rss_type as usize].add_on_cpu(cpu_id, val);
161        self.rss_hwm_counters[rss_type as usize]
162            .get_on_cpu(cpu_id)
163            .fetch_max(rss, Ordering::Relaxed);
164    }
165}

167/// The type representing categories of Resident Set Size (RSS).
168///
169/// Reference: <https://elixir.bootlin.com/linux/v6.16.5/source/include/linux/mm_types_task.h#L26-L32>
170#[repr(u32)]
171#[derive(Clone, Copy, Debug, Eq, PartialEq, TryFromInt)]
172pub(crate) enum RssType {
173    File = 0,
174    Anon = 1,
175}
176
177const NUM_RSS_COUNTERS: usize = 2;
178
179pub(super) struct RssDelta<'a> {
180    delta: [isize; NUM_RSS_COUNTERS],
181    operated_vmar: &'a Vmar,
182}
183
184#[derive(Clone, Copy, Debug, Projectable, serde::Serialize)]
185pub(crate) struct RssDeltaChangeEvent {
186    increment: isize,
187    current: usize,
188}
189
190impl<'a> RssDelta<'a> {
191    pub(super) fn new(operated_vmar: &'a Vmar) -> Self {
192        Self {
193            delta: [0; NUM_RSS_COUNTERS],
194            operated_vmar,
195        }
196    }
197
198    pub(super) fn add(&mut self, rss_type: RssType, increment: isize) {
199        #[cfg(not(baseline_asterinas))]
200        if rss_type == RssType::Anon {
201            let current = if increment > 0 {
202                oqueues::GLOBAL_RSS.fetch_add(increment as usize, Ordering::Relaxed)
203                    + increment as usize
204            } else {
205                // The `-increment` is to make conversions to usize correct.
206                oqueues::GLOBAL_RSS.fetch_sub(-increment as usize, Ordering::Relaxed)
207                    - (-increment) as usize
208            };
209
210            trace_structured_data!(
211                { mm.rss_delta.anon },
212                RssDeltaChangeEvent { increment, current },
213                RssDeltaChangeEvent
214            )
215        }
216
217        self.delta[rss_type as usize] += increment;
218    }
219
220    fn get(&self, rss_type: RssType) -> isize {
221        self.delta[rss_type as usize]
222    }
223}
224
225impl Drop for RssDelta<'_> {
226    fn drop(&mut self) {
227        for i in 0..NUM_RSS_COUNTERS {
228            let rss_type = RssType::try_from(i as u32).unwrap();
229            let delta = self.get(rss_type);
230            self.operated_vmar.add_rss_counter(rss_type, delta);
231        }
232    }
233}

235/// A holder for a reverse mapping entry that is to be removed.
236///
237/// This holds the [`Vmo`] object, so the lock guard for reverse mappings can
238/// have a correct lifetime (see [`Self::remove`]).
239#[must_use]
240struct RmapToRemove {
241    vmo: Option<Arc<Vmo>>,
242}
243
244impl RmapToRemove {
245    pub(self) fn new(vmo: Option<Arc<Vmo>>) -> Self {
246        Self { vmo }
247    }
248
249    pub(self) fn remove(&mut self, vmar: &Vmar, addr: Vaddr) -> Option<MutexGuard<'_, Rmap>> {
250        let mut rmap = self.vmo.as_ref()?.rmap().lock();
251        rmap.remove(vmar.weak_self.clone(), addr);
252        Some(rmap)
253    }
254}

256struct VmarInner {
257    /// The mapped pages and associated metadata.
258    ///
259    /// When inserting a `VmMapping` into this set, use `insert_try_merge` to
260    /// auto-merge adjacent and compatible mappings, or `insert_without_try_merge`
261    /// if the mapping is known not mergeable with any neighboring mappings.
262    vm_mappings: IntervalSet<Vaddr, VmMapping>,
263    /// The total mapped memory in bytes.
264    total_vm: usize,
265}
266
267impl VmarInner {
268    const fn new() -> Self {
269        Self {
270            vm_mappings: IntervalSet::new(),
271            total_vm: 0,
272        }
273    }

275    /// Returns `Ok` if the calling process may expand its mapped
276    /// memory by the passed size.
277    fn check_extra_size_fits_rlimit(&self, expand_size: usize) -> Result<()> {
278        let Some(process) = Process::current() else {
279            // When building a `Process`, the kernel task needs to build
280            // some `VmMapping`s, in which case this branch is reachable.
281            return Ok(());
282        };
283
284        let rlimit_as = process
285            .resource_limits()
286            .get_rlimit(ResourceType::RLIMIT_AS)
287            .get_cur();
288
289        if rlimit_as.saturating_sub(self.total_vm as u64) < expand_size as u64 {
290            return_errno_with_message!(Errno::ENOMEM, "the address space size limit is reached");
291        }
292        Ok(())
293    }
294
295    /// Checks whether `addr..addr + size` is covered by a single `VmMapping`,
296    /// and returns the address of the single `VmMapping` if successful.
297    fn check_lies_in_single_mapping(&self, addr: Vaddr, size: usize) -> Result<Vaddr> {
298        if let Some(vm_mapping) = self
299            .vm_mappings
300            .find_one(&addr)
301            .filter(|vm_mapping| vm_mapping.map_end() - addr >= size)
302        {
303            Ok(vm_mapping.map_to_addr())
304        } else {
305            return_errno_with_message!(Errno::EFAULT, "the range must lie in a single mapping");
306        }
307    }

309    /// Inserts a `VmMapping` into the `Vmar`, without attempting to merge with
310    /// neighboring mappings.
311    ///
312    /// The caller must ensure that the given `VmMapping` is not mergeable with
313    /// any neighboring mappings.
314    ///
315    /// Make sure the insertion doesn't exceed address space limit.
316    fn insert_without_try_merge(
317        &mut self,
318        vmar: &Vmar,
319        vm_mapping: VmMapping,
320        rmap: Option<&mut Rmap>,
321    ) {
322        self.total_vm += vm_mapping.map_size();
323
324        if let Some(rmap) = rmap {
325            rmap.insert(
326                vmar.weak_self.clone(),
327                RmapEntry {
328                    vaddr: vm_mapping.map_to_addr(),
329                    offset: vm_mapping.vmo().unwrap().offset(),
330                    size: vm_mapping.map_size(),
331                },
332            );
333        }
334        self.vm_mappings.insert(vm_mapping);
335    }
336
337    /// Inserts a `VmMapping` into the `Vmar`, and attempts to merge it with
338    /// neighboring mappings.
339    ///
340    /// This method will try to merge the `VmMapping` with neighboring mappings
341    /// that are adjacent and compatible, in order to reduce fragmentation.
342    ///
343    /// Make sure the insertion doesn't exceed address space limit.
344    fn insert_try_merge(
345        &mut self,
346        vmar: &Vmar,
347        vm_mapping: VmMapping,
348        mut rmap: Option<&mut Rmap>,
349    ) {
350        self.total_vm += vm_mapping.map_size();
351        let mut vm_mapping = vm_mapping;
352        let addr = vm_mapping.map_to_addr();
353
354        if let Some(prev) = self.vm_mappings.find_prev(&addr) {
355            let (new_mapping, to_remove) = vm_mapping.try_merge_with(prev);
356            vm_mapping = new_mapping;
357            if let Some(addr) = to_remove {
358                self.vm_mappings.remove(&addr);
359                if let Some(rmap) = rmap.as_deref_mut() {
360                    rmap.remove(vmar.weak_self.clone(), addr);
361                }
362            }
363        }
364
365        if let Some(next) = self.vm_mappings.find_next(&addr) {
366            let (new_mapping, to_remove) = vm_mapping.try_merge_with(next);
367            vm_mapping = new_mapping;
368            if let Some(addr) = to_remove {
369                self.vm_mappings.remove(&addr);
370                if let Some(rmap) = rmap.as_deref_mut() {
371                    rmap.remove(vmar.weak_self.clone(), addr);
372                }
373            }
374        }
375
376        if let Some(rmap) = rmap {
377            rmap.insert(
378                vmar.weak_self.clone(),
379                RmapEntry {
380                    vaddr: vm_mapping.map_to_addr(),
381                    offset: vm_mapping.vmo().unwrap().offset(),
382                    size: vm_mapping.map_size(),
383                },
384            );
385        }
386        self.vm_mappings.insert(vm_mapping);
387    }
388
389    /// Removes a `VmMapping` based on the provided key from the `Vmar`.
390    fn remove(&mut self, key: &Vaddr) -> Option<(VmMapping, RmapToRemove)> {
391        let vm_mapping = self.vm_mappings.remove(key)?;
392        self.total_vm -= vm_mapping.map_size();
393
394        let rmap_to_remove = RmapToRemove::new(vm_mapping.vmo_for_rmap().cloned());
395
396        Some((vm_mapping, rmap_to_remove))
397    }

399    /// Finds a set of [`VmMapping`]s that intersect with the provided range.
400    fn query(&self, range: &Range<Vaddr>) -> impl Iterator<Item = &VmMapping> {
401        self.vm_mappings.find(range)
402    }

404    /// Calculates the total amount of overlap between `VmMapping`s
405    /// and the provided range.
406    fn count_overlap_size(&self, range: Range<Vaddr>) -> usize {
407        let mut sum_overlap_size = 0;
408        for vm_mapping in self.vm_mappings.find(&range) {
409            let vm_mapping_range = vm_mapping.range();
410            let intersected_range = get_intersected_range(&range, &vm_mapping_range);
411            sum_overlap_size += intersected_range.end - intersected_range.start;
412        }
413        sum_overlap_size
414    }

416    /// Allocates a free region for mapping with a specific offset and size.
417    ///
418    /// If the provided range is already occupied, return an error.
419    fn alloc_free_region_exact(&mut self, offset: Vaddr, size: usize) -> Result<Range<Vaddr>> {
420        if self
421            .vm_mappings
422            .find(&(offset..offset + size))
423            .next()
424            .is_some()
425        {
426            return_errno_with_message!(
427                Errno::EEXIST,
428                "the range contains pages that are already mapped"
429            );
430        }
431
432        Ok(offset..(offset + size))
433    }

435    /// Allocates a free region for mapping with a specific offset and size.
436    ///
437    /// If the provided range is already occupied, this function truncates all
438    /// the mappings that intersect with the range.
439    fn alloc_free_region_exact_truncate(
440        &mut self,
441        vmar: &Vmar,
442        offset: Vaddr,
443        size: usize,
444        rss_delta: &mut RssDelta,
445    ) -> Range<Vaddr> {
446        let range = offset..offset + size;
447
448        let mut mappings_to_remove = Vec::new();
449        for vm_mapping in self.vm_mappings.find(&range) {
450            mappings_to_remove.push(vm_mapping.map_to_addr());
451        }
452
453        for vm_mapping_addr in mappings_to_remove {
454            let (vm_mapping, mut rmap_to_remove) = self.remove(&vm_mapping_addr).unwrap();
455            let mut rmap = rmap_to_remove.remove(vmar, vm_mapping_addr);
456
457            let vm_mapping_range = vm_mapping.range();
458            let intersected_range = get_intersected_range(&range, &vm_mapping_range);
459
460            let (left, taken, right) = vm_mapping.split_range(&intersected_range);
461            if let Some(left) = left {
462                self.insert_without_try_merge(vmar, left, rmap.as_deref_mut());
463            }
464            if let Some(right) = right {
465                self.insert_without_try_merge(vmar, right, rmap.as_deref_mut());
466            }
467
468            // Note that `rmap` must be dropped before `taken`. Otherwise,
469            // there is a possibility of a deadlock because a device mapping
470            // may attempt to access its reverse mappings in `Drop`.
471            drop(rmap);
472
473            rss_delta.add(taken.rss_type(), -(taken.unmap(&vmar.vm_space) as isize));
474        }
475
476        range
477    }

479    /// Allocates a free region for mapping, searching from high address to low address.
480    ///
481    /// If no such region is found, return an error.
482    fn alloc_free_region(&mut self, size: usize, align: usize) -> Result<Range<Vaddr>> {
483        // This value represents the highest possible address for a new mapping.
484        // For simplicity, we use a fixed value `2048` here. The value contains the following considerations:
485        // - The stack fixed padding size.
486        // - The stack random padding size.
487        // - The future growth of the stack.
488        // FIXME: This value should consider the process's actual stack configuration, which may
489        // exist in `ResourceLimits`.
490        let high_limit = VMAR_CAP_ADDR - INIT_STACK_SIZE - PAGE_SIZE * 2048;
491        let low_limit = VMAR_LOWEST_ADDR;
492        self.alloc_free_region_in_range(size, align, low_limit, high_limit)
493    }
494
495    /// Allocates a free region for mapping that resides below 2 GiB.
496    ///
497    /// This is specifically used for supporting the `MAP_32BIT` mmap flag.
498    #[cfg(target_arch = "x86_64")]
499    fn alloc_free_region_below_2gib(&mut self, size: usize, align: usize) -> Result<Range<Vaddr>> {
500        let high_limit = MAP_32BIT_HIGH_LIMIT.min(VMAR_CAP_ADDR);
501        let low_limit = VMAR_LOWEST_ADDR;
502        self.alloc_free_region_in_range(size, align, low_limit, high_limit)
503    }

505    /// Core logic for `alloc_free_region` and `alloc_free_region_below_2gib`.
506    ///
507    /// Searches for a free region in `[low_limit, high_limit)`, from high to low.
508    fn alloc_free_region_in_range(
509        &mut self,
510        size: usize,
511        align: usize,
512        low_limit: Vaddr,
513        high_limit: Vaddr,
514    ) -> Result<Range<Vaddr>> {
515        fn try_alloc_in_hole(
516            hole_start: Vaddr,
517            hole_end: Vaddr,
518            size: usize,
519            align: usize,
520        ) -> Option<Range<Vaddr>> {
521            let start = hole_end.checked_sub(size)?.align_down(align);
522            if start >= hole_start {
523                Some(start..start + size)
524            } else {
525                None
526            }
527        }
528
529        let mut prev_vm_mapping_start = high_limit;
530        for vm_mapping in self.vm_mappings.iter().rev() {
531            let hole_start = vm_mapping.range().end.max(low_limit);
532            let hole_end = prev_vm_mapping_start.min(high_limit);
533
534            if let Some(region) = try_alloc_in_hole(hole_start, hole_end, size, align) {
535                return Ok(region);
536            }
537
538            prev_vm_mapping_start = vm_mapping.range().start;
539            if prev_vm_mapping_start <= low_limit {
540                break;
541            }
542        }
543
544        // Check the hole between `low_limit` and the lowest mapping.
545        if prev_vm_mapping_start > low_limit {
546            let hole_start = low_limit;
547            let hole_end = prev_vm_mapping_start.min(high_limit);
548            if let Some(region) = try_alloc_in_hole(hole_start, hole_end, size, align) {
549                return Ok(region);
550            }
551        }
552
553        return_errno_with_message!(Errno::ENOMEM, "no free region for mapping can be found");
554    }

556    /// Splits and unmaps the found mapping if the new size is smaller.
557    /// Enlarges the last mapping if the new size is larger.
558    fn resize_mapping(
559        &mut self,
560        vmar: &Vmar,
561        map_addr: Vaddr,
562        old_size: usize,
563        new_size: usize,
564        rss_delta: &mut RssDelta,
565    ) -> Result<()> {
566        debug_assert_eq!(map_addr % PAGE_SIZE, 0);
567        debug_assert_eq!(old_size % PAGE_SIZE, 0);
568        debug_assert_eq!(new_size % PAGE_SIZE, 0);
569
570        if new_size == 0 {
571            return_errno_with_message!(Errno::EINVAL, "resizing a mapping to zero is invalid");
572        }
573
574        if new_size == old_size {
575            return Ok(());
576        }
577
578        if !is_userspace_vaddr_range(map_addr, new_size) {
579            return_errno_with_message!(Errno::EINVAL, "the address range is not in userspace");
580        }
581
582        if new_size < old_size {
583            // Shrink the mapping. The old mapping is larger.
584            if map_addr.checked_add(old_size).is_none() {
585                return_errno_with_message!(Errno::EINVAL, "the address range overflows");
586            }
587            self.alloc_free_region_exact_truncate(
588                vmar,
589                map_addr + new_size,
590                old_size - new_size,
591                rss_delta,
592            );
593            return Ok(());
594        }
595
596        // Expand the mapping. The old mapping is smaller.
597        let old_map_end = map_addr + old_size;
598        self.alloc_free_region_exact(old_map_end, new_size - old_size)
599            .map_err(|_| {
600                Error::with_message(
601                    Errno::ENOMEM,
602                    "the range contains pages that are already mapped",
603                )
604            })?;
605
606        let last_mapping = self.vm_mappings.find_one(&(old_map_end - 1)).unwrap();
607        let last_mapping_addr = last_mapping.map_to_addr();
608        debug_assert_eq!(last_mapping.map_end(), old_map_end);
609
610        if !last_mapping.can_expand() {
611            return_errno_with_message!(Errno::EFAULT, "device mappings cannot be expanded");
612        }
613
614        self.check_extra_size_fits_rlimit(new_size - old_size)?;
615        let (last_mapping, mut rmap_to_remove) = self.remove(&last_mapping_addr).unwrap();
616        let mut rmap = rmap_to_remove.remove(vmar, last_mapping_addr);
617        let last_mapping = last_mapping.enlarge(new_size - old_size);
618        self.insert_try_merge(vmar, last_mapping, rmap.as_deref_mut());
619        Ok(())
620    }
