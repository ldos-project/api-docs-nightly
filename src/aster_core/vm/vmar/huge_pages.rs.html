3use core::sync::atomic::{AtomicBool, Ordering};
4
5use ostd::mm::{
6    PagingConsts, PagingLevel, page_size,
7    vm_space::{VmMappingPolicy, VmMappingRequest},
8};
9
10/// Enable hugepage usage using some default policy.
11static MAP_HUGE_ENABLED: AtomicBool = AtomicBool::new(false);
12aster_cmdline::define_flag_param!("vm.huge_mapping_enabled", MAP_HUGE_ENABLED);
13
14/// Returns true if huge pages were enabled on the kernel CLI
15pub fn huge_mapping_enabled() -> bool {
16    MAP_HUGE_ENABLED.load(Ordering::Relaxed)
17}
18
19/// Attempt to preserve hugepage mapping when the user application advises the kernel it no longer
20/// needs pages.
21static MAP_HUGE_PRESERVE_ON_DONTNEED: AtomicBool = AtomicBool::new(false);
22aster_cmdline::define_flag_param!(
23    "vm.huge_mapping_preserve_on_dontneed",
24    MAP_HUGE_PRESERVE_ON_DONTNEED
25);
26
27/// Returns true if huge page mappings should be preserved when a MADV_DONTNEED is issued.
28#[expect(unused)]
29pub(crate) fn huge_mapping_preserve_on_dontneed() -> bool {
30    MAP_HUGE_PRESERVE_ON_DONTNEED.load(Ordering::Relaxed)
31}
32
33/// VmMappingPolicy implementation that always maps a huge page when possible.
34pub(super) struct VmMappingPolicyGreedyHugeMapping {}
35
36impl VmMappingPolicy for VmMappingPolicyGreedyHugeMapping {
37    fn get_page_level(&self, req: &VmMappingRequest) -> PagingLevel {
38        // Check if the address is aligned to a level 2 page. If it is not aligned, it cannot be
39        // mapped at a level larger than 1.
40        if req
41            .page_aligned_addr
42            .is_multiple_of(page_size::<PagingConsts>(2))
43        {
44            2
45        } else {
46            1
47        }
48    }
49}
