3//! Regression-test support for the OQFS file operations (`produce`, `consume`, `strong_observe`,
4//! and `weak_observe`).
5//!
6//! When enabled via the `oqfs_test.enable` kernel command-line flag, this module registers three
7//! OQueues under the `oqfs_test` OQFS subtree and spawns a single kernel thread. That thread blocks
8//! on the `trigger` queue and, each time userspace writes to `trigger/produce`, produces the batch
9//! `0..BATCH` into the `observe` queue, which userspace reads through `strong_observe` or
10//! `weak_observe`. The `userspace_produce_consume` queue is produced and consumed entirely in
11//! userspace; the kernel only keeps it alive.
12//!
13//! The userspace drivers under `test/initramfs/src/regression/oqfs/` drive the queues through the
14//! `/oqueues/oqfs_test/...` filesystem and verify the values they observe.
15
16use core::sync::atomic::{AtomicBool, Ordering};
17
18use ostd::{
19    orpc::oqueue::{
20        ConsumableOQueue, ConsumableOQueueRef, Consumer, OQueueBase, ValueProducer, registry,
21    },
22    path,
23};
24
25use crate::thread::kernel_thread::ThreadOptions;
26
27/// Master switch (`oqfs_test.enable`); the module is inert unless set.
28static ENABLE: AtomicBool = AtomicBool::new(false);
29aster_cmdline::define_flag_param!("oqfs_test.enable", ENABLE);
30
31/// Number of values produced into the observe queue per trigger.
32const BATCH: u32 = 16;
33
34/// Capacity of the `userspace_produce_consume` and `observe` queues.
35const CAPACITY: usize = 64;
36
37/// The prepared OQFS test endpoints.
38struct OqfsTest {
39    /// Strong handle keeping the `userspace_produce_consume` queue alive; it is produced and consumed entirely in
40    /// userspace and never read by the kernel.
41    #[expect(
42        dead_code,
43        reason = "held only to keep the userspace_produce_consume OQueue (and its OQFS export) alive"
44    )]
45    userspace_produce_consume: ConsumableOQueueRef<u32>,
46    observe_producer: ValueProducer<u32>,
47    trigger_consumer: Consumer<u32>,
48}
49
50impl OqfsTest {
51    /// Creates and exports the three OQueues on the `oqfs_test` OQFS subtree, or returns `None`
52    /// if `oqfs_test.enable` is not set.
53    fn prepare() -> Option<Self> {
54        if !ENABLE.load(Ordering::Relaxed) {
55            return None;
56        }
57
58        // The `userspace_produce_consume` queue is produced and consumed entirely in userspace; the kernel keeps only
59        // the strong reference so the queue (and its OQFS export) stays alive.
60        let userspace_produce_consume_path = path!(oqfs_test.userspace_produce_consume);
61        let userspace_produce_consume =
62            ConsumableOQueueRef::<u32>::new(CAPACITY, userspace_produce_consume_path.clone());
63        registry::register_producible(&userspace_produce_consume_path, &userspace_produce_consume);
64
65        // The `observe` queue is observable: the kernel produces into it and userspace reads it
66        // through `strong_observe` or `weak_observe`.
67        let observe_path = path!(oqfs_test.observe);
68        let observe = ConsumableOQueueRef::<u32>::new(CAPACITY, observe_path.clone());
69        registry::register(&observe_path, &observe.as_any_oqueue());
70        let observe_producer = observe
71            .attach_value_producer()
72            .expect("the oqfs_test observe OQueue always allows a value producer");
73
74        // The `trigger` queue is produced by userspace and consumed by the test thread.
75        let trigger_path = path!(oqfs_test.trigger);
76        let trigger = ConsumableOQueueRef::<u32>::new(4, trigger_path.clone());
77        registry::register_producible(&trigger_path, &trigger);
78        let trigger_consumer = trigger
79            .attach_consumer()
80            .expect("the oqfs_test trigger OQueue always allows attaching its consumer");
81
82        Some(Self {
83            userspace_produce_consume,
84            observe_producer,
85            trigger_consumer,
86        })
87    }
88
89    /// The test thread body: block on the trigger queue and produce the batch on each trigger.
90    fn run(self) {
91        loop {
92            // Block (without spinning) until userspace writes to the trigger queue.
93            let _ = self.trigger_consumer.consume();
94            // Produce the batch that userspace observes through strong/weak observation.
95            for value in 0..BATCH {
96                self.observe_producer.produce(value);
97            }
98        }
99    }
100}
101
102/// Starts the OQFS test endpoints on their own kernel thread.
103pub(crate) fn init_after_init_process() {
104    let Some(test) = OqfsTest::prepare() else {
105        return;
106    };
107    ThreadOptions::new(move || test.run()).spawn();
108}
